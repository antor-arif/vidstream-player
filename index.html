Installation

# NPM
npm install vidstream-player

# Yarn
yarn add vidstream-player

# PNPM
pnpm add vidstream-player

Quick Start (HTML)

<!-- Include the SDK -->
<script src="https://unpkg.com/vidstream-player/dist/vidstream-player.umd.js"></script>

<!-- Add the player -->
<video-player
  src="https://example.com/video.m3u8"
  poster="https://example.com/poster.jpg"
  mode="ott"
  accent-color="#e50914"
></video-player>

JavaScript API

const player = document.getElementById('my-player');

// Playback controls
player.play();
player.pause();
player.togglePlay();
player.seek(30); // Seek to 30 seconds

// Volume & playback
player.setVolume(0.5);
player.toggleMute();
player.setPlaybackRate(1.5);

// Fullscreen & PiP
player.toggleFullscreen();
player.togglePip();

// Get current state
const state = player.getState();
console.log(state.currentTime, state.duration);

// Set chapters
player.setChapters([
  { id: '1', title: 'Introduction', startTime: 0 },
  { id: '2', title: 'Main Content', startTime: 60 },
  { id: '3', title: 'Conclusion', startTime: 180 }
]);

// Event listeners
player.addEventListener('player-play', () => console.log('Playing'));
player.addEventListener('player-pause', () => console.log('Paused'));
player.addEventListener('player-ended', () => console.log('Ended'));
player.addEventListener('player-timeupdate', (e) => {
  console.log('Time:', e.detail.state.currentTime);
});

NPM / ES Modules

// Import the web component
import 'vidstream-player';

// Or create programmatically
import { VidstreamPlayer } from 'vidstream-player';

const player = VidstreamPlayer.create('#container', {
  source: {
    src: 'https://example.com/video.m3u8',
    type: 'hls',
    poster: 'https://example.com/poster.jpg'
  },
  mode: 'ott',
  theme: {
    accentColor: '#e50914'
  },
  watermark: {
    enabled: true,
    text: 'My Brand',
    position: 'top-right'
  }
});
